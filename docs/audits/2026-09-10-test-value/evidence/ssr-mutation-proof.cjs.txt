function waitForMessage(child) {
  return new Promise((resolve, reject) => {
    const timer = setTimeout(() => reject(new Error('message timeout')), 10000);
    child.once('message', message => { clearTimeout(timer); resolve(message); });
    child.once('error', reject);
    child.once('exit', code => { clearTimeout(timer); reject(new Error(`child exited: ${code}`)); });
  });
}

async function reservePort() {
  const server = http.createServer();
  await new Promise(resolve => server.listen(0, '127.0.0.1', resolve));
  const port = server.address().port;
  await new Promise(resolve => server.close(resolve));
  return port;
}

function checkSsrGuard() {
  const req = createRequire(path.join(installedRepo, 'packages/runtime/plugin-runtime/package.json'));
  const { transformSync } = req('esbuild');
  const React = req('react');
  const { renderToString } = req('react-dom/server');
  const filename = path.join(snapshot, 'packages/runtime/plugin-runtime/src/boundary-debugger/index.tsx');
  const original = fs.readFileSync(filename, 'utf8');
  const guard = '  if (!mounted) {\n    return null;\n  }';
  if (original.split(guard).length !== 2) throw new Error('SSR guard is not unique');
  const mutant = original.replace(guard, '');
  function render(source, controlMode) {
    const code = transformSync(source, { loader: 'tsx', format: 'cjs', jsx: 'automatic' }).code;
    const module = { exports: {} };
    new Function('require', 'module', 'exports', code)(req, module, module.exports);
    let WrappedApp;
    module.exports.default({
      controlMode,
      enabledByDefault: true,
      legacySelector: '[data-mf-remote]',
      metadata: { appId: 'shell', boundaries: [{ appId: 'catalog', mfName: 'catalog' }], schemaVersion: 1 },
    }).setup({ wrapRoot(factory) { WrappedApp = factory(() => React.createElement('main', null, 'app')); } });
    return renderToString(React.createElement(WrappedApp));
  }
  return ['hidden', 'visible'].map(controlMode => {
    const originalHtml = render(original, controlMode);
    const mutantHtml = render(mutant, controlMode);
    return {
      controlMode,
      originalHtml,
      mutantHtml,
      expectedHtml: '<main>app</main>',
      currentOracleSurvived: mutantHtml === '<main>app</main>',
    };
  });
}
